#include <SPI.h>
#include <MFRC522.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// Konfiguracja pinów RFID RC522
constexpr uint8_t RST_PIN = 9;
constexpr uint8_t SS_PIN = 10;
MFRC522 mfrc522(SS_PIN, RST_PIN);

// Konfiguracja wyświetlacza LCD
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Dane kart
const String UID_STUDENT = "F2 7B DF 00";
const String NAZWA_STUDENTA = "Jan Kowalski";

const String UID_KSIAZKA = "60 7E B3 56";
const String TYTUL_KSIAZKI = "Podrecznik";

// Zmienne stanu
bool czyWypozyczona = false;
String wypozyczonaPrzez = "";
bool czekamNaKsiazke = false;

void setup() {
  Serial.begin(9600);
  
  // 1. Inicjalizacja ekranu LCD
  Wire.begin();
  lcd.init();
  lcd.backlight();
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Test czytnika...");

  // 2. Wymuszenie stabilnego stanu na pinach SPI (ważne dla Arduino R4!)
  pinMode(SS_PIN, OUTPUT);
  digitalWrite(SS_PIN, HIGH);
  pinMode(RST_PIN, OUTPUT);
  digitalWrite(RST_PIN, HIGH);
  delay(50);

  // 3. Inicjalizacja SPI i czytnika RFID
  SPI.begin();
  mfrc522.PCD_Init();
  delay(100);
  
  // Wzmocnienie sygnału anteny do maksimum (pomaga przy słabszym zasilaniu)
  mfrc522.PCD_SetAntennaGain(mfrc522.RxGain_max);

  // 4. Sprawdzenie, czy Arduino w ogóle widzi czytnik RC522 po kablach
  byte version = mfrc522.PCD_ReadRegister(mfrc522.VersionReg);
  lcd.setCursor(0, 1);
  if (version == 0x00 || version == 0xFF) {
    lcd.print("BLAD Kysli/Kabli");
    Serial.println("BLAD: Czytnik RC522 nie odpowiada! Sprawdz przewody.");
    while (true); // Zatrzymaj program, dopóki kable nie będą poprawione
  } else {
    lcd.print("Czytnik OK!");
    Serial.print("Czytnik wykryty, wersja: 0x");
    Serial.println(version, HEX);
    delay(1500);
  }
  
  pokazEkranStartowy();
}

void loop() {
  // Sprawdzanie, czy przyłożono nową kartę
  if (!mfrc522.PICC_IsNewCardPresent()) {
    return;
  }
  if (!mfrc522.PICC_ReadCardSerial()) {
    return;
  }

  // Odczytanie UID przyłożonej karty
  String odczytaneUID = pobierzUID();
  delay(1500);
  Serial.println("Zeskanowano UID: " + odczytaneUID);

  // KROK 1: Oczekiwanie na skan legitymacji studenta
  if (!czekamNaKsiazke) {
    if (odczytaneUID == UID_STUDENT) {
      czekamNaKsiazke = true;
      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("Witaj, Jan!");
      lcd.setCursor(0, 1);
      lcd.print("Skanuj ksiazke..");
    } 
    else if (odczytaneUID == UID_KSIAZKA) {
      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("Blad! Najpierw");
      lcd.setCursor(0, 1);
      lcd.print("skanuj legitym.!");
      delay(2500);
      pokazEkranStartowy();
    } 
    else {
      nieznanaKarta(odczytaneUID);
    }
  } 
  // KROK 2: Legitymacja zeskanowana -> oczekiwanie na książkę
  else {
    if (odczytaneUID == UID_KSIAZKA) {
      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print(TYTUL_KSIAZKI);
      lcd.setCursor(0, 1);
      
      if (!czyWypozyczona) {
        czyWypozyczona = true;
        wypozyczonaPrzez = NAZWA_STUDENTA;
        lcd.print("Wypozyczono!");
      } else {
        czyWypozyczona = false;
        wypozyczonaPrzez = "";
        lcd.print("Zwrocono!");
      }
      
      delay(3000);
      czekamNaKsiazke = false;
      pokazEkranStartowy();
    } 
    else if (odczytaneUID == UID_STUDENT) {
      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("Anulowano");
      lcd.setCursor(0, 1);
      lcd.print("operacje.");
      delay(2000);
      czekamNaKsiazke = false;
      pokazEkranStartowy();
    } 
    else {
      nieznanaKarta(odczytaneUID);
    }
  }

  // Zatrzymanie odczytu tej samej karty i reset komunikacji
  mfrc522.PICC_HaltA();
  mfrc522.PCD_StopCrypto1();
  delay(500); // Krótka przerwa zapobiegająca podwójnemu odczytowi
}

String pobierzUID() {
  String uidString = "";
  for (byte i = 0; i < mfrc522.uid.size; i++) {
    if (mfrc522.uid.uidByte[i] < 0x10) {
      uidString += "0";
    }
    uidString += String(mfrc522.uid.uidByte[i], HEX);
    if (i < mfrc522.uid.size - 1) {
      uidString += " ";
    }
  }
  uidString.toUpperCase();
  return uidString;
}

void pokazEkranStartowy() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("1.Zeskanuj swoja");
  lcd.setCursor(0, 1);
  lcd.print("legitymacje...");
}

void nieznanaKarta(String uid) {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Nieznana karta:");
  lcd.setCursor(0, 1);
  lcd.print(uid); // Wyświetli UID nieznanej karty na ekranie!
  delay(2500);
  if (czekamNaKsiazke) {
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Witaj, Jan!");
    lcd.setCursor(0, 1);
    lcd.print("Skanuj ksiazke..");
  } else {
    pokazEkranStartowy();
  }
}
